\section{Ejercicio con matemáticas}

\begin{ejercicio}[Ecuación cuadrática]
Resolver la ecuación $x^2 + 5x + 6 = 0$ usando la fórmula general.
\end{ejercicio}

\begin{resolucion}
Aplicando la fórmula general:
\[
    x = \frac{-b \pm \sqrt{b^2 - 4ac}}{2a}
\]

Donde $a = 1$, $b = 5$, $c = 6$:
\[
    x = \frac{-5 \pm \sqrt{25 - 24}}{2} = \frac{-5 \pm 1}{2}
\]

Por lo tanto:
\[
    x_1 = \frac{-5 + 1}{2} = -2, \quad x_2 = \frac{-5 - 1}{2} = -3
\]

Las soluciones son $x_1 = -2$ y $x_2 = -3$.
\end{resolucion}

\section{Ejercicio con código}

\begin{ejercicio}
Escribir una función en Python que calcule la media aritmética de una lista.
\end{ejercicio}

\begin{resolucion}

\begin{lstlisting}[style=python, caption={Función para calcular la media}]
def media(valores: list[float]) -> float:
    """Devuelve la media aritmética de una lista de valores."""
    if not valores:
        raise ValueError("La lista no puede estar vacía.")
    return sum(valores) / len(valores)

# Ejemplo de uso
numeros = [1, 2, 3, 4, 5]
resultado = media(numeros)
print(f"La media de {numeros} es: {resultado}")  # 3.0
\end{lstlisting}

\end{resolucion}

\section{Ejercicio con teorema}

\begin{teorema}[Teorema de Pitágoras]
En un triángulo rectángulo, el cuadrado de la hipotenusa es igual a la suma de los cuadrados de los catetos.
\[
    c^2 = a^2 + b^2
\]
\end{teorema}

\begin{definicion}[Triángulo rectángulo]
Un triángulo rectángulo es aquel que tiene un ángulo de 90 grados.
\end{definicion}

\begin{ejemplo}
Si los catetos de un triángulo rectángulo miden 3 y 4 unidades, entonces la hipotenusa mide:
\[
    c = \sqrt{3^2 + 4^2} = \sqrt{9 + 16} = \sqrt{25} = 5 \text{ unidades}
\]
\end{ejemplo}
