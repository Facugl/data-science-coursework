\section{Ejercicio 2}

\begin{ejercicio}
Escribir una función en Python que calcule la media de una lista.
\end{ejercicio}

\begin{resolucion}

\begin{lstlisting}[style=python, caption={Media aritmética en Python}]
def media(valores: list[float]) -> float:
    """Devuelve la media aritmética de una lista de valores."""
    if not valores:
        raise ValueError("La lista no puede estar vacía.")
    return sum(valores) / len(valores)

print(media([1, 2, 3, 4, 5]))  # 3.0
\end{lstlisting}

\end{resolucion}
