\section{Desarrollar la expresión de cada notación de sumatoria}

\begin{enumerate}[label=\alph*.-]
    \item $\displaystyle\sum_{i=1}^{5}(2i-1)$

    \solucion
    \[
    \sum_{i=1}^{5}(2i-1) = (2\cdot1-1)+(2\cdot2-1)+(2\cdot3-1)+(2\cdot4-1)+(2\cdot5-1)
    = 1+3+5+7+9 = 25
    \]

    \item $\displaystyle\sum_{i=2}^{6}\frac{i+2}{1-i}$

    \solucion
    \[
    \sum_{i=2}^{6}\frac{i+2}{1-i}
    = \frac{2+2}{1-2}+\frac{3+2}{1-3}+\frac{4+2}{1-4}+\frac{5+2}{1-5}+\frac{6+2}{1-6}
    = -4-\frac{5}{2}-2-\frac{7}{4}-\frac{8}{5} = -\frac{237}{20}
    \]

    \item $\displaystyle\sum_{i=1}^{6}\frac{(-1)^{i}}{1+i}$

    \solucion
    \[
    \sum_{i=1}^{6}\frac{(-1)^{i}}{1+i}
    = \frac{(-1)^1}{1+1}+\frac{(-1)^2}{1+2}+\frac{(-1)^3}{1+3}+\frac{(-1)^4}{1+4}+\frac{(-1)^5}{1+5}+\frac{(-1)^6}{1+6}
    \]
    \[
    = -\frac{1}{2}+\frac{1}{3}-\frac{1}{4}+\frac{1}{5}-\frac{1}{6}+\frac{1}{7} \approx -0{,}24
    \]

    \item $\displaystyle\sum_{i=5}^{9}\left(\frac{1}{2}i-3\right)$

    \solucion
    \[
    \sum_{i=5}^{9}\left(\frac{1}{2}i-3\right)
    = \left(\frac{5}{2}-3\right)+\left(\frac{6}{2}-3\right)+\left(\frac{7}{2}-3\right)+\left(\frac{8}{2}-3\right)+\left(\frac{9}{2}-3\right)
    \]
    \[
    = -15 + \frac{35}{2} = \frac{5}{2}
    \]

    \newpage
    \item $\displaystyle\sum_{i=4}^{10}\frac{n}{2i}$

    \solucion
    \[
    \sum_{i=4}^{10}\frac{n}{2i}
    = \frac{n}{2}\sum_{i=4}^{10}\frac{1}{i}
    = \frac{n}{2}\left(\frac{1}{4}+\frac{1}{5}+\frac{1}{6}+\frac{1}{7}+\frac{1}{8}+\frac{1}{9}+\frac{1}{10}\right)
    = \frac{n}{2}\cdot\frac{2761}{2520}
    = \frac{2761\,n}{5040}
    \]

    \item $\displaystyle\sum_{i=1}^{6}8$

    \solucion
    \[
    \sum_{i=1}^{6}8 = 8\sum_{i=1}^{6}1 = 8\cdot6 = 48
    \]

\end{enumerate}

\section{Aplicar las propiedades del operador sumatoria para calcular}

\begin{enumerate}[label=\alph*.-]
    \item $\displaystyle\sum_{k=4}^{12}(5k-2)$

    \solucion
    \[
    \sum_{k=4}^{12}(5k-2) = 5\sum_{k=4}^{12}k - 2\sum_{k=4}^{12}1
    \]
    \[
    \sum_{k=4}^{12}k = \frac{(4+12)\cdot9}{2} = 72
    \qquad\qquad
    \sum_{k=4}^{12}1 = 9
    \]
    \[
    = 5\cdot72 - 2\cdot9 = 360 - 18 = 342
    \]

    \item $\displaystyle\sum_{k=1}^{6}(4k+3)^2$

    \solucion
    \[
    \sum_{k=1}^{6}(4k+3)^2 = \sum_{k=1}^{6}(16k^2+24k+9)
    = 16\sum_{k=1}^{6}k^2 + 24\sum_{k=1}^{6}k + 9\sum_{k=1}^{6}1
    \]
    \[
    = 16\cdot\frac{6(6+1)(2\cdot6+1)}{6} + 24\cdot\frac{(6+1)}{2} + 9\cdot6
    = 16(7\cdot13) + 24(3\cdot7) + 54
    \]
    \[
    = 1456 + 504 + 54 = 2014
    \]

    \item $\displaystyle\sum_{h=3}^{8}(h+4)h$

    \solucion
    \[
    \sum_{h=3}^{8}(h+4)h = \sum_{h=3}^{8}(h^2+4h)
    = \sum_{h=3}^{8}h^2 + 4\sum_{h=3}^{8}h
    \]
    \[
    \sum_{h=3}^{8}h^2
    = \sum_{h=1}^{8}h^2 - \sum_{h=1}^{2}h^2
    = \frac{8\cdot9\cdot17}{6} - \frac{2\cdot3\cdot5}{6}
    = 204 - 5 = 199
    \]
    \[
    \sum_{h=3}^{8}h
    = \frac{(3+8)\cdot6}{2} = 33
    \]
    \[
    = 199 + 4\cdot33 = 199 + 132 = 331
    \]

    \item $\displaystyle\sum_{h=1}^{n}(2^h - 2^{h-1})$

    \solucion
    \[
    \sum_{h=1}^{n}(2^h - 2^{h-1}) = \sum_{h=1}^{n}2^h - \sum_{h=1}^{n}2^{h-1}
    = (2^1+2^2+\cdots+2^n) - (2^0+2^1+\cdots+2^{n-1})
    = 2^n - 2^0 = 2^n - 1
    \]

\end{enumerate}

\section{Completar las líneas de puntos para que se verifiquen las igualdades}

\begin{enumerate}[label=\alph*.-]
    \item $\displaystyle\sum_{i=1}^{12} 4^i = \sum_{i=6}^{\ldots} \ldots$

    \solucion

    Aplicando la propiedad $\displaystyle\sum_{k=1}^{n}2k = \sum_{k=p+1}^{n+p}2(k-p)$:
    \[
    \sum_{i=1}^{12} 4^i = \sum_{i=1+5}^{12+5} 4^{i-5} = \sum_{i=6}^{17} 4^{i-5}
    \]

    \item $\displaystyle\sum_{j=1}^{7}(3j-2) = \sum_{i=6}^{\ldots} \ldots$

    \solucion
    \[
    \sum_{j=1}^{7}(3j-2) = \sum_{j=1+5}^{7+5}\bigl[3(j-5)-2\bigr] = \sum_{j=6}^{12}(3j-17)
    \]

    \item $\displaystyle\sum_{k=8}^{15}(3k+1)k = \sum_{k=1}^{\ldots} \ldots$

    \solucion

    Aplicando el cambio $k \leftarrow k+7$ (es decir, $k_{\text{nuevo}} = k_{\text{viejo}}-7$):
    cuando $k=8 \Rightarrow k'=1$ y cuando $k=15 \Rightarrow k'=8$:
    \[
    \sum_{k=8}^{15}(3k+1)k
    = \sum_{k=1}^{8}\bigl(3(k+7)+1\bigr)(k+7)
    = \sum_{k=1}^{8}(3k+22)(k+7)
    \]

\end{enumerate}

\section{Determinar si las siguientes igualdades son verdaderas o falsas. Justificar la respuesta}

\begin{enumerate}[label=\alph*.-]
    \item $\displaystyle\sum_{i=-p}^{p}\frac{i}{p} = \frac{1}{p}$

    \solucion

    Desarrollando la sumatoria:
    \[
    \sum_{i=-p}^{p}\frac{i}{p} = \frac{1}{p}\sum_{i=-p}^{p}i
    = \frac{1}{p}\left(-p + \cdots + (-1) + 0 + 1 + \cdots + p\right)
    = \frac{1}{p}\cdot 0 = 0 \neq \frac{1}{p}
    \]
    Los términos se cancelan simétricamente: $-\dfrac{p}{p}+\dfrac{p}{p} = -1+1 = 0$.
    Por lo tanto la igualdad es \textbf{FALSA}.

    \item $\displaystyle\sum_{i=1}^{p} i = \frac{p(p+1)}{2}$

    \solucion

    La igualdad es \textbf{VERDADERA}. Demostración por inducción matemática:

    \textbf{Caso base} ($p=1$):
    \[
    \sum_{i=1}^{1} i = 1 = \frac{1\cdot2}{2} = 1 \quad \checkmark
    \]

    \textbf{Hipótesis inductiva}: supongamos que para $p=k$ la igualdad vale:
    \[
    \sum_{i=1}^{k} i = \frac{k(k+1)}{2}
    \]

    \textbf{Paso inductivo}: debemos probar que vale para $p=k+1$, es decir, que
    $\displaystyle\sum_{i=1}^{k+1} i = \frac{(k+1)(k+2)}{2}$.
    \[
    \sum_{i=1}^{k+1} i = \sum_{i=1}^{k} i + (k+1)
    \overset{H.I.}{=} \frac{k(k+1)}{2} + (k+1)
    = (k+1)\left(\frac{k}{2}+1\right)
    = \frac{(k+1)(k+2)}{2} \quad \checkmark
    \]

    \newpage
    \item $\displaystyle\sum_{i=1}^{p} i^2 = \frac{p(p+1)(2p+1)}{6}$

    \solucion

    La igualdad es \textbf{VERDADERA}. Demostración por inducción matemática:

    \textbf{Caso base} ($p=1$):
    \[
    \sum_{i=1}^{1} i^2 = 1 = \frac{1\cdot2\cdot3}{6} = 1 \quad \checkmark
    \]

    \textbf{Hipótesis inductiva}: supongamos que para $p=k$ la igualdad vale:
    \[
    \sum_{i=1}^{k} i^2 = \frac{k(k+1)(2k+1)}{6}
    \]

    \textbf{Paso inductivo}: debemos probar que vale para $p=k+1$, es decir, que
    $\displaystyle\sum_{i=1}^{k+1} i^2 = \frac{(k+1)(k+2)(2k+3)}{6}$.
    \[
    \sum_{i=1}^{k+1} i^2 = \sum_{i=1}^{k} i^2 + (k+1)^2
    \overset{H.I.}{=} \frac{k(k+1)(2k+1)}{6} + (k+1)^2
    = (k+1)\left[\frac{k(2k+1)}{6} + (k+1)\right]
    \]
    \[
    = (k+1)\cdot\frac{k(2k+1)+6(k+1)}{6}
    = (k+1)\cdot\frac{2k^2+7k+6}{6}
    = \frac{(k+1)(k+2)(2k+3)}{6} \quad \checkmark
    \]

    \newpage
    \item $\displaystyle\sum_{i=1}^{p} i^3 = \frac{p^2(p+1)^2}{4}$

    \solucion

    La igualdad es \textbf{VERDADERA}. Demostración por inducción matemática:

    \textbf{Caso base} ($p=1$):
    \[
    \sum_{i=1}^{1} i^3 = 1 = \frac{1^2\cdot2^2}{4} = 1 \quad \checkmark
    \]

    \textbf{Hipótesis inductiva}: supongamos que para $p=k$ la igualdad vale:
    \[
    \sum_{i=1}^{k} i^3 = \frac{k^2(k+1)^2}{4}
    \]

    \textbf{Paso inductivo}: debemos probar que vale para $p=k+1$, es decir, que
    $\displaystyle\sum_{i=1}^{k+1} i^3 = \frac{(k+1)^2(k+2)^2}{4}$.
    \[
    \sum_{i=1}^{k+1} i^3 = \sum_{i=1}^{k} i^3 + (k+1)^3
    \overset{H.I.}{=} \frac{k^2(k+1)^2}{4} + (k+1)^3
    = (k+1)^2\left[\frac{k^2}{4} + (k+1)\right]
    \]
    \[
    = (k+1)^2\cdot\frac{k^2+4k+4}{4}
    = \frac{(k+1)^2(k+2)^2}{4} \quad \checkmark
    \]

\end{enumerate}

\section{Expresar los siguientes productos utilizando el operador correspondiente}

\begin{enumerate}[label=\alph*.-]
    \item $3 \cdot 13 \cdot 18 \cdot 23 \cdot 28$

    \solucion

    Observando que la razón entre los dos primeros terminos es el doble del resto, asumo que falta un termino intermedio, luego, la sucesión $3,\,8,\,13,\,18,\,23,\,28$ es una progresión aritmética de primer término $3$ y razón $5$, con término general $5k-2$ para $k=1,2,\ldots,6$:
    \[
    k=1:\;3\;\checkmark,\quad k=2:\;8\;\checkmark,\quad k=3:\;13\;\checkmark,\quad
    k=4:\;18\;\checkmark,\quad k=5:\;23\;\checkmark,\quad k=6:\;28\;\checkmark
    \]
    \[
    \prod_{k=1}^{6}(5k-2)
    \]

    \item $\left(-\dfrac{1}{9}\right)\cdot\dfrac{1}{3}\cdot(-1)\cdot3\cdot9\cdot27$

    \solucion

    Los valores absolutos son $3^{-2},\,3^{-1},\,3^{0},\,3^{1},\,3^{2},\,3^{3}$,
    es decir potencias de $3$ con exponente $k$ para $k=-2,-1,0,1,2,3$.
    Los signos siguen el patrón $(-1)^{k+1}$:
    \[
    k=-2:\;(-1)^{-1}\cdot3^{-2}=-\tfrac{1}{9}\;\checkmark,\quad
    k=-1:\;(-1)^{0}\cdot3^{-1}=+\tfrac{1}{3}\;\checkmark,\quad
    k=0:\;(-1)^{1}\cdot1=-1\;\checkmark,\;\ldots
    \]
    \[
    \prod_{k=-2}^{3}(-1)^{k+1}\cdot 3^{k}
    \]

    \item $1\cdot4\cdot27\cdot256\cdot3125$

    \solucion

    Reconocemos que $1=1^1,\;4=2^2,\;27=3^3,\;256=4^4,\;3125=5^5$,
    es decir cada factor es de la forma $k^k$:
    \[
    \prod_{k=1}^{5}k^{k}
    \]

    \newpage
    \item $\left(-\dfrac{1}{7}\right)\cdot\dfrac{1}{21}\cdot\left(-\dfrac{1}{63}\right)\cdot\dfrac{1}{189}\cdot\left(-\dfrac{1}{567}\right)$

    \solucion

    Los denominadores $7,\,21,\,63,\,189,\,567$ forman una progresión geométrica de razón $3$:
    $7\cdot3^{k-1}$ para $k=1,2,3,4,5$.
    Los signos alternan comenzando en negativo: $(-1)^{k}$ para $k=1,2,3,4,5$.
    \[

\section{Resolver y calcular}

\begin{enumerate}[label=\alph*.-]
    \item $\dfrac{(2+3)!}{2!\cdot3!}$

    \solucion
    \[
    \frac{(2+3)!}{2!\cdot3!} = \frac{5!}{2!\cdot3!} = \frac{5\cdot4\cdot3!}{2!\cdot3!}
    = \frac{5\cdot4}{2} = \frac{20}{2} = 10
    \]

    \item $\dfrac{(n+5)!}{(n+3)!}$

    \solucion
    \[
    \frac{(n+5)!}{(n+3)!} = \frac{(n+5)(n+4)(n+3)!}{(n+3)!}
    = (n+5)(n+4) = n^2+9n+20
    \]

    \item $\dfrac{(n-1)!}{n!}+\dfrac{(n+1)!}{n!}$

    \solucion
    \[
    \frac{(n-1)!}{n!}+\frac{(n+1)!}{n!}
    = \frac{(n-1)!}{n(n-1)!}+\frac{(n+1)\cdot n!}{n!}
    = \frac{1}{n}+(n+1)
    = \frac{1}{n}+n+1
    = \frac{n^2+n+1}{n}
    \]

    \item $\dbinom{7}{5}$

    \solucion
    \[
    \binom{7}{5} = \frac{7!}{(7-5)!\,5!} = \frac{7!}{2!\,5!}
    = \frac{7\cdot6\cdot5!}{2!\,5!} = \frac{7\cdot6}{2} = 21
    \]

    \item $\dbinom{11}{1}$

    \solucion
    \[
    \binom{11}{1} = \frac{11!}{(11-1)!\,1!} = \frac{11\cdot10!}{10!\cdot1} = 11
    \]

    \item $\dbinom{11}{10}$

    \solucion
    \[
    \binom{11}{10} = \frac{11!}{(11-10)!\,10!} = \frac{11\cdot10!}{1!\,10!} = 11
    \]

    \newpage
    \item $\dbinom{n+5}{n+4}$

\end{enumerate}

\section{Binomio de Newton}

\begin{enumerate}[label=\alph*.-]
    \item Dar la expresión genérica del Binomio de Newton, usarla para realizar el desarrollo de $\left(\dfrac{x}{3}-2\right)^7$. ¿Cuántos sumandos tiene este desarrollo?

    \solucion

    La expresión genérica del Binomio de Newton es:
    \[
    (a+b)^n = \sum_{k=0}^{n}\binom{n}{k}a^{n-k}b^k
    \]

    Con $a=\dfrac{x}{3}$, $b=-2$ y $n=7$:
    \[
    \left(\frac{x}{3}-2\right)^7 = \sum_{k=0}^{7}\binom{7}{k}\left(\frac{x}{3}\right)^{7-k}(-2)^k
    \]
    \[
    = \binom{7}{0}\left(\frac{x}{3}\right)^7(-2)^0
    + \binom{7}{1}\left(\frac{x}{3}\right)^6(-2)^1
    + \binom{7}{2}\left(\frac{x}{3}\right)^5(-2)^2
    + \binom{7}{3}\left(\frac{x}{3}\right)^4(-2)^3
    \]
    \[
    + \binom{7}{4}\left(\frac{x}{3}\right)^3(-2)^4
    + \binom{7}{5}\left(\frac{x}{3}\right)^2(-2)^5
    + \binom{7}{6}\left(\frac{x}{3}\right)^1(-2)^6
    + \binom{7}{7}\left(\frac{x}{3}\right)^0(-2)^7
    \]

    Se observa que hay $n+1$ términos $\Rightarrow 7+1 = \mathbf{8}$ \textbf{términos}.

    \item Escribir la expresión del término genérico. En base a ello, encontrar $T_5$.

    \solucion

    El término genérico (término $k+1$) es:
    \[
    T_{k+1} = \binom{n}{k}\,a^{n-k}\,b^k
    \]

    Para $T_5$ tomamos $k=4$:
    \[
    T_5 = \binom{7}{4}\left(\frac{x}{3}\right)^{7-4}(-2)^4
    = \binom{7}{4}\left(\frac{x}{3}\right)^3 \cdot 16
    \]
    \[
    \binom{7}{4} = \frac{7!}{4!\,(7-4)!} = \frac{7\cdot6\cdot5\cdot4!}{4!\cdot3!}
    = \frac{7\cdot6\cdot5}{6} = 35
    \]
    \[
    T_5 = 35\cdot\frac{x^3}{27}\cdot16 = \frac{560}{27}\,x^3
    \]

    \newpage
    \item Explicar cómo encontrar el o los términos centrales. ¿Cuál es el término central de $\left(\dfrac{x}{3}-2\right)^7$?

    \solucion

    Para un desarrollo de $n+1$ términos:
    \begin{itemize}
        \item Si $n$ es \textbf{par}: hay un único término central, $T_{\frac{n}{2}+1}$.
        \item Si $n$ es \textbf{impar}: hay \textbf{dos} términos centrales, $T_{\frac{n+1}{2}}$ y $T_{\frac{n+3}{2}}$.
    \end{itemize}

    Como $n=7$ es impar, hay dos términos centrales: $T_{\frac{7+1}{2}} = T_4$ y
    $T_{\frac{7+3}{2}} = T_5$.

    \textbf{Término $T_4$} ($k=3$):
    \[
    T_4 = \binom{7}{3}\left(\frac{x}{3}\right)^4(-2)^3
    = 35\cdot\frac{x^4}{81}\cdot(-8)
    = 35\left(\frac{x}{3}\right)^4\!(-8)
    = -\frac{280}{81}{x^4}
    \]

    \textbf{Término $T_5$} ($k=4$):
    \[
    T_5 = \binom{7}{4}\left(\frac{x}{3}\right)^3(-2)^4
    = 35\left(\frac{x}{3}\right)^3\cdot16
    = \frac{560}{27}{x^3}
    \]

    Donde $\dbinom{7}{3} = \dbinom{7}{4} = \dfrac{7!}{3!\,4!} = \dfrac{7\cdot6\cdot5\cdot4!}{6\cdot4!} = 35$.

\end{enumerate}

    \]

    Como $n=10$ es par, hay un único término central: $T_{\frac{n}{2}+1} = T_6$.

    Con $a=\dfrac{1}{3}x$, $b=5$, $n=10$ y $k=5$:

    T_6 = \binom{10}{5}\left(\frac{x}{3}\right)^5(5)^5

    = 6\cdot y\cdot(-32x^5)
    = -192\,yx^5
    \]
    \[

    \item El término independiente del desarrollo de $\left(p-\dfrac{1}{p}\right)^8$.

    \]
