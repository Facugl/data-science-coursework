% ============================================================
% COMANDO CODIGOJAVA
% Muestra el codigo Java desde un archivo .java
% Uso: \codigojava{ruta/al/archivo/Solucion.java}
% ============================================================
\newcommand{\codigojava}[1]{%
    \vspace{1em}%
    \textbf{C\'odigo Java:}%
    \vspace{0.5em}%
    \lstinputlisting[style=java]{#1}%
}

% ============================================================
% COMANDO CODIGOJAVA MULTIPLE
% Para mostrar múltiples archivos Java (clase + ejecutable)
% Uso: \codigojavaclase{Persona.java}{MainApp.java}
% ============================================================
\newcommand{\codigojavaclase}[2]{%
    \vspace{1em}%
    \textbf{Clase Persona:}%
    \vspace{0.5em}%
    \lstinputlisting[style=java]{#1}%
    \vspace{1em}%
    \textbf{Clase Ejecutable:}%
    \vspace{0.5em}%
    \lstinputlisting[style=java]{#2}%
}

% ============================================================

    \vspace{0.5em}%
    \lstinputlisting[style=java]{#3}%
}

% ============================================================
% COMANDO CAPTURA
% Muestra la captura de pantalla con la salida del programa
% Uso: \captura{ruta/al/archivo/captura.png}
% ============================================================
\newcommand{\captura}[1]{%
    \vspace{1em}%
    \textbf{Salida del programa:}%
    \vspace{0.5em}%
    \begin{figure}[H]%
        \centering%
        \includegraphics[width=0.9\textwidth]{#1}%
    \end{figure}%
}

% ============================================================
% COMANDO CAPTURA CON ANCHO PERSONALIZADO
% Uso: \capturacustom{ruta/al/archivo/captura.png}{0.5}
% El segundo parámetro es el ancho (0.5 = 50%, 1.0 = 100%)
% ============================================================
\newcommand{\capturacustom}[2]{%
    \vspace{1em}%
    \textbf{Salida del programa:}%
    \vspace{0.5em}%
    \begin{figure}[H]%
        \centering%
        \includegraphics[width=#2\textwidth]{#1}%
    \end{figure}%
}
