\definecolor{codebg}{RGB}{248, 248, 248}
\definecolor{codeframe}{RGB}{200, 200, 200}
\definecolor{codecomment}{RGB}{106, 153, 85}
\definecolor{codekeyword}{RGB}{86, 156, 214}
\definecolor{codestring}{RGB}{206, 145, 120}

% ============================================================
% PALABRAS CLAVE DE PSEINT
% ============================================================
\lstdefinelanguage{PSeInt}{
    morekeywords={
        Algoritmo, FinAlgoritmo,
        Definir, Como, Entero, Real, Caracter, Logico, Cadena,
        Leer, Escribir,
        Si, Entonces, SiNo, FinSi,
        Segun, De, FinSegun,
        Mientras, Hacer, FinMientras,
        Repetir, HastaQue,
        Para, Hasta, Con, Paso, FinPara,
        Y, O, No, Mod, Div,
        Verdadero, Falso
    },
    sensitive=false,
    morecomment=[l]{//},
    morestring=[b]",
    morestring=[b]',
}

% ============================================================
% ESTILO PSEINT
% NOTA: inputencoding=latin1 es necesario porque PSeInt en
% Windows guarda los .psc en codificacion cp1252/latin1.
% Sin esto, las tildes y la enie aparecen como caracteres
% incorrectos en el PDF compilado.
% ============================================================
\lstdefinestyle{pseint}{
    language=PSeInt,
    inputencoding=latin1,
    extendedchars=true,
    backgroundcolor=\color{codebg},
    frame=single,
    rulecolor=\color{codeframe},
    basicstyle=\ttfamily\small,
    commentstyle=\color{codecomment}\itshape,
    keywordstyle=\color{codekeyword}\bfseries,
    stringstyle=\color{codestring},
    numberstyle=\tiny\color{codenumber},
    numbers=left,
    numbersep=8pt,
    breaklines=true,
    breakatwhitespace=true,
    tabsize=4,
    showstringspaces=false,
    captionpos=b,
    literate=
        {á}{{\'{a}}}1 {é}{{\'{e}}}1 {í}{{\'{i}}}1 {ó}{{\'{o}}}1 {ú}{{\'{u}}}1
        {Á}{{\'{A}}}1 {É}{{\'{E}}}1 {Í}{{\'{I}}}1 {Ó}{{\'{O}}}1 {Ú}{{\'{U}}}1
        {ñ}{{\~{n}}}1 {Ñ}{{\~{N}}}1 {ü}{{\"u}}1 {¿}{{?`}}1 {¡}{{!`}}1,
}
