\begin{enumerate}[label=\roman*.]
    % ======================== item i ========================
    \item $\displaystyle\lim_{x \to 1} f(x)$

    \solucion

    En la gráfica vemos que los valores de $f(x)$ tienden a $2$ conforme $x$ tiende a $1$. Los límites laterales por izquierda y derecha se aproximan a $2$. Entonces:
    \[
        \lim_{x \to 1} f(x) = 2
    \]

    % ======================== item ii ========================
    \item $\displaystyle\lim_{x \to 3^-} f(x)$

    \solucion

    Observando la gráfica de $f(x)$, cuando $x$ tiende a $3$ por izquierda, $f(x)$ tiende a ser $1$:
    \[
        \lim_{x \to 3^-} f(x) = 1
    \]

    % ======================== item iii ========================
    \item $\displaystyle\lim_{x \to 3^+} f(x)$

    \solucion

    De manera similar, cuando $x$ tiende a $3$ pero por derecha, se observa que $f(x)$ tiende a $4$:
    \[
        \lim_{x \to 3^+} f(x) = 4
    \]

    \newpage
    % ======================== item iv ========================
    \item $\displaystyle\lim_{x \to 3} f(x)$

    \solucion

    Al evaluar el límite de $x$ tendiendo a $3$, se tiene:
    \[
        \lim_{x \to 3^-} f(x) \neq \lim_{x \to 3^+} f(x)
    \]
    Al tener los límites laterales por izquierda y derecha con valores distintos, podemos concluir que \textbf{no existe} el límite de $f(x)$ cuando $x \to 3$.

    % ======================== item v ========================
    \item $f(3)$

    \solucion

    Observando la gráfica se tiene que cuando $x = 3$:
    \[
        f(3) = 3
    \]
\end{enumerate}

\section[Evaluación de límites]{Evalúe el límite y justifique cada paso indicando las leyes de los límites apropiadas.}

Recordemos las siguientes leyes de los límites. Sean $f(x)$ y $g(x)$ funciones tales que $\lim_{x\to a} f(x) = L_1$ y $\lim_{x\to a} g(x) = L_2$, y sea $c$ un número real. Entonces:

\begin{enumerate}[label=\arabic*.-]
    \item $\displaystyle\lim_{x\to a}[f(x)+g(x)] = \lim_{x\to a}f(x) + \lim_{x\to a}g(x) = L_1 + L_2$
    \item $\displaystyle\lim_{x\to a}[f(x)-g(x)] = \lim_{x\to a}f(x) - \lim_{x\to a}g(x) = L_1 + L_2$
    \item $\displaystyle\lim_{x\to a}[c\cdot f(x)] = c\cdot\lim_{x\to a}f(x) = c\cdot L_1$
    \item $\displaystyle\lim_{x\to a}[f(x)\cdot g(x)] = \lim_{x\to a}f(x)\cdot\lim_{x\to a}g(x) = L_1\cdot L_2$
    \item $\displaystyle\lim_{x\to a}\frac{f(x)}{g(x)} = \frac{\lim_{x\to a}f(x)}{\lim_{x\to a}g(x)} = \frac{L_1}{L_2}$, siempre que $L_2 \neq 0$
    \item $\displaystyle\lim_{x\to a}[f(x)]^n = \left[\lim_{x\to a}f(x)\right]^n$, donde $n$ es un entero $>0$
    \item $\displaystyle\lim_{x\to a} c = c$
    \item $\displaystyle\lim_{x\to a} x = a$
    \item $\displaystyle\lim_{x\to a} x^n = a^n$
    \item $\displaystyle\lim_{x\to a} \sqrt[n]{x} = \sqrt[n]{a}$, con $n$ un entero $>0$
    \item $\displaystyle\lim_{x\to a} \sqrt[n]{f(x)} = \sqrt[n]{\lim_{x\to a}f(x)}$
\end{enumerate}

\begin{enumerate}[label=\alph*)]
    % ======================== item a ========================
    \item $\lim_{x \to 3} (5x^3 - 3x^2 + x - 6)$

    \solucion
    \begin{align*}
        \lim_{x \to 3}(5x^3 - 3x^2 + x - 6)
        &= \lim_{x\to3}(5x^3) - \lim_{x\to3}(3x^2) + \lim_{x\to3}x - \lim_{x\to3}6 && \text{(leyes 1 y 2)}\\
        &= 5\lim_{x\to3}x^3 - \lim_{x\to3}x - 3\lim_{x\to3}x^2 - \lim_{x\to3}6 && \text{(ley 3)}\\
        &= 5\cdot 3^3 - 3\cdot 3^2 - 3 - 6 && \text{(leyes 7, 8 y 9)}\\
        &= 5\cdot 27 - 3\cdot 9 - 3 - 6\\
        &= 135 - 27 - 9 = 99
    \end{align*}
    \[
        \therefore\quad \lim_{x \to 3}(5x^3 - 3x^2 + x - 6) = 99
    \]

    \newpage
    % ======================== item b ========================
    \item $\lim_{x \to 1} (1 + \sqrt[3]{8})(2 - 6x^2 + x^3)$

    \solucion
    \begin{align*}
        \lim_{x \to 1}(1+\sqrt[3]{8})(2-6x^2+x^3)
        &= (1+\sqrt[3]{8}) \cdot \lim_{x\to1}(2-6x^2+x^3) && \text{(ley 3)}\\
        &= (1+\sqrt[3]{8})\left[\lim_{x\to1}2 - \lim_{x\to1}6x^2 + \lim_{x\to1}x^3\right] && \text{(leyes 1 y 2)}\\
        &= (1+2)\left[2 - 6\cdot1^2 + 1^3\right] && \text{(leyes 7, 9 y 10)}\\
        &= 3\left[2 - 6 + 1\right] = 3(-3) = -9
    \end{align*}
    \[
        \therefore\quad \lim_{x\to1}(1+\sqrt[3]{8})(2-6x^2+x^3) = -9
    \]

    % ======================== item c ========================
    \item $\lim_{t \to 2} \left(\dfrac{t^2 - 2}{t^3 - 3t + 5}\right)^2$

    \solucion
    \begin{align*}
        \lim_{t\to2}\left(\frac{t^2-2}{t^3-3t+5}\right)^2
        &= \left[\lim_{t\to2}\left(\frac{t^2-2}{t^3-3t+5}\right)\right]^2 && \text{(ley 6)}\\
        &= \left[\frac{\lim_{t\to2}(t^2-2)}{\lim_{t\to2}(t^3-3t+5)}\right]^2 && \text{(ley 5)}\\
        &= \left[\frac{\lim_{t\to2}t^2 - \lim_{t\to2}2}{\lim_{t\to2}t^3 - \lim_{t\to2}3t + \lim_{t\to2}5}\right]^2 && \text{(leyes 1 y 2)}\\
        &= \left[\frac{\lim_{t\to2}t^2 - \lim_{t\to2}2}{\lim_{t\to2}t^3 - 3\lim_{t\to2}t + \lim_{t\to2}5}\right]^2 && \text{(ley 3)}\\
        &= \left[\frac{2^2 - 2}{2^3 - 3\cdot2 + 5}\right]^2 && \text{(leyes 7, 8 y 9)}\\
        &= \left[\frac{4-2}{8-6+5}\right]^2 = \left(\frac{2}{7}\right)^2 = \frac{4}{49}
    \end{align*}
    \[
        \therefore\quad \lim_{t\to2}\left(\frac{t^2-2}{t^3-3t+5}\right)^2 = \frac{4}{49}
    \]

    \newpage
    % ======================== item d ========================
    \item $\lim_{x \to 2}\sqrt{\dfrac{2x^2 + 1}{3x - 2}}$

    \solucion
    \begin{align*}
        \lim_{x\to2}\sqrt{\frac{2x^2+1}{3x-2}}
        &= \sqrt{\lim_{x\to2}\frac{2x^2+1}{3x-2}} && \text{(ley 11)}\\
        &= \sqrt{\frac{\lim_{x\to2}(2x^2+1)}{\lim_{x\to2}(3x-2)}} && \text{(ley 5)}\\
        &= \sqrt{\frac{\lim_{x\to2}(2x^2) + \lim_{x\to2}1}{\lim_{x\to2}(3x) - \lim_{x\to2}2}} && \text{(leyes 1 y 2)}\\
        &= \sqrt{\frac{2\lim_{x\to2}x^2 + \lim_{x\to2}1}{3\lim_{x\to2}x - \lim_{x\to2}2}} && \text{(ley 3)}\\
        &= \sqrt{\frac{2\cdot2^2 + 1}{3\cdot2 - 2}} && \text{(leyes 7, 8 y 9)}\\
        &= \sqrt{\frac{9}{4}} = \frac{\sqrt{9}}{\sqrt{4}} = \frac{3}{2}
    \end{align*}
    \[
        \therefore\quad \lim_{x\to2}\sqrt{\frac{2x^2+1}{3x-2}} = \frac{3}{2}
    \]
\end{enumerate}

\section[Límites que requieren simplificación]{Encuentre cada uno de los siguientes límites si éstos existen. Si el límite no existe, explique por qué.}

\begin{enumerate}[label=\alph*)]
    % ======================== item a ========================
    \item $\lim_{x \to 3} (2x + |x - 3|)$

    \solucion

    Recordamos que:
    \[
        |x-3| = \begin{cases} x-3 & \text{si } x \geq 3 \\ -(x-3) & \text{si } x < 3 \end{cases}
    \]

    \textbf{Límite por derecha} ($x > 3$, luego $|x-3| = x-3$):
    \begin{align*}
        \lim_{x\to3^+}(2x+|x-3|) &= \lim_{x\to3^+}(2x+x-3) = 2\cdot3+3-3 = 6
    \end{align*}

    \textbf{Límite por izquierda} ($x < 3$, luego $|x-3| = 3-x$):
    \begin{align*}
        \lim_{x\to3^-}(2x+|x-3|) &= \lim_{x\to3^-}(2x+3-x) = 2\cdot3+3-3 = 6
    \end{align*}

    Debido a que el valor de los límites laterales coinciden:
    \[
        \lim_{x\to3}(2x+|x-3|) = 6
    \]

    % ======================== item b ========================
    \item $\lim_{x \to -6} \dfrac{2x + 12}{|x + 6|}$

    \solucion

    Recordamos que:
    \[
        |x+6| = \begin{cases} x+6 & \text{si } x > -6 \\ -(x+6) & \text{si } x < -6 \end{cases}
    \]

    \textbf{Límite por derecha} ($x > -6$, luego $|x+6| = x+6$):
    \[
        \lim_{x\to-6^+}\frac{2x+12}{x+6} = \lim_{x\to-6^+}\frac{2(x+6)}{x+6} = 2
    \]

    \textbf{Límite por izquierda} ($x < -6$, luego $|x+6| = -(x+6)$):
    \[
        \lim_{x\to-6^-}\frac{2x+12}{-x-6} = \lim_{x\to-6^-}\frac{2(x+6)}{-(x+6)} = \frac{2}{-1} = -2
    \]

    Luego podemos concluir que el límite \textbf{no existe}:
    \[
        \lim_{x\to-6^-}\frac{2x+12}{|x+6|} \neq \lim_{x\to-6^+}\frac{2x+12}{|x+6|}
    \]

    % ======================== item c ========================
    \item $\lim_{x \to 2} \left(\dfrac{1}{x} - \dfrac{1}{|x|}\right)$

    \solucion

    Analizamos el límite por izquierda y por derecha. Para $x > 2$ se tiene $|x| = x$, entonces:
    \[
        \lim_{x\to2^+}\left(\frac{1}{x}-\frac{1}{|x|}\right) = \lim_{x\to2^+}\left(\frac{1}{x}-\frac{1}{x}\right) = \lim_{x\to2^+}0 = 0
    \]

    Para $x < 2$ también $|x| = x$ (ya que $x > 0$), entonces:
    \[
        \lim_{x\to2^-}\left(\frac{1}{x}-\frac{1}{|x|}\right) = \lim_{x\to2^-}\left(\frac{1}{x}-\frac{1}{x}\right) = \lim_{x\to2^-}0 = 0
    \]

    Luego el límite existe:
    \[
        \lim_{x\to2}\left(\frac{1}{x}-\frac{1}{|x|}\right) = 0
    \]
\end{enumerate}

\section[Límites de funciones definidas por partes]{Resuelva lo pedido en cada caso.}

\begin{enumerate}[label=\alph*)]
    % ======================== item a ========================
    \item Sea $f(x) = \begin{cases} (x^2 + 1) & \text{si } x < 1 \\ (x - 2)^2 & \text{si } x \geq 1 \end{cases}$

    \begin{enumerate}[label=\roman*.]
        \item Encuentre $\lim_{x \to 1^-} f(x)$ y $\lim_{x \to 1^+} f(x)$.

        \solucion

        Cuando analizamos $x \to 1^-$ se tiene $x < 1$:
        \[
            \lim_{x\to1^-}(x^2+1) = 1^2+1 = 2
        \]

        Mientras que analizando el límite por derecha se tiene $x \geq 1$:
        \[
            \lim_{x\to1^+}(x-2)^2 = (1-2)^2 = (-1)^2 = 1
        \]

        \item ¿Existe $\lim_{x \to 1} f(x)$?

        \solucion

        \textbf{No existe} $\displaystyle\lim_{x\to1}f(x)$ porque:
        \[

\section[Límites algebraicos]{Evalúe cada uno de los siguientes límites si éstos existen.}

\begin{enumerate}[label=\alph*)]
    % ======================== item a ========================
    \item $\lim_{x \to 5} \dfrac{x^2 - 6x + 5}{x - 5}$

    \solucion

    No podemos encontrar el límite por sustitución directa. Intentamos factorizar el numerador como una diferencia de cuadrados:
    \begin{align*}
        x^2 - 6x &= -5 \\
        x^2 - 6x + 9 &= -5 + 9 \\
        (x-3)^2 &= 4 \implies x - 3 = \pm 2 \implies x_1 = 5,\quad x_2 = 1
    \end{align*}
    Entonces el límite nos queda:
    \[
        \lim_{x\to5}\frac{(x-5)(x-1)}{x-5} = \lim_{x\to5}(x-1) = 4
    \]
    \[
        \therefore\quad \lim_{x\to5}\frac{x^2-6x+5}{x-5} = 4
    \]

    % ======================== item b ========================
    \item $\lim_{t \to 1} \dfrac{t^4-1}{2t^3-1}$

    \solucion
    \begin{align*}
        \lim_{t\to1}\frac{t^4-1}{t^3-1}
        &= \lim_{t\to1}\frac{(t^2-1)(t^2+1)}{(t-1)(t^2+t+1)}
        = \lim_{t\to1}\frac{(t-1)(t+1)(t^2+1)}{(t-1)(t^2+t+1)} \\
        &= \lim_{t\to1}\frac{(t+1)(t^2+1)}{t^2+t+1}
        = \frac{2 \cdot 2}{1+1+1} = \frac{4}{3}
    \end{align*}
    \[
        \therefore\quad \lim_{t\to1}\frac{t^4-1}{t^3-1} = \frac{4}{3}
    \]

    % ======================== item c ========================
    \item $\lim_{h \to 0} \dfrac{(h - 5)^2 - 25}{h}$

    \solucion

    Factorizamos el numerador:
    \begin{align*}
        (h-5)^2 - 25 &= h^2 - 10h + 25 - 25 = h^2 - 10h = h(h-10)
    \end{align*}
    \[
        \lim_{h\to0}\frac{h(h-10)}{h} = \lim_{h\to0}(h-10) = -10
    \]
    \[
        \therefore\quad \lim_{h\to0}\frac{(h-5)^2-25}{h} = -10
    \]

    % ======================== item d ========================
    \item $\lim_{h \to 0} \dfrac{\sqrt{9 + h} - 3}{h}$

    \solucion

    Racionalizamos el numerador:
    \begin{align*}
        \frac{\sqrt{9+h}-3}{h} \cdot \frac{\sqrt{9+h}+3}{\sqrt{9+h}+3}
        &= \frac{9+h-9}{h(\sqrt{9+h}+3)}
        = \frac{h}{h(\sqrt{9+h}+3)}
        = \frac{1}{\sqrt{9+h}+3}
    \end{align*}
    \[
        \lim_{h\to0}\frac{1}{\sqrt{9+h}+3} = \frac{1}{\sqrt{9+0}+3} = \frac{1}{3+3} = \frac{1}{6}
    \]
    \[
        \therefore\quad \lim_{h\to0}\frac{\sqrt{9+h}-3}{h} = \frac{1}{6}
    \]

    % ======================== item e ========================
    \item $\lim_{u \to 2} \dfrac{\sqrt{4u+1}-3}{u-2}$

    \solucion

    Racionalizamos el numerador:
    \begin{align*}
        \frac{\sqrt{4u+1}-3}{u-2} \cdot \frac{\sqrt{4u+1}+3}{\sqrt{4u+1}+3}
        &= \frac{4u+1-9}{(u-2)(\sqrt{4u+1}+3)}
        = \frac{4u-8}{(u-2)(\sqrt{4u+1}+3)} \\
        &= \frac{4(u-2)}{(u-2)(\sqrt{4u+1}+3)}
        = \frac{4}{\sqrt{4u+1}+3}
    \end{align*}
    \[
        \lim_{u\to2}\frac{4}{\sqrt{4u+1}+3} = \frac{4}{\sqrt{8+1}+3} = \frac{4}{\sqrt{9}+3} = \frac{4}{6} = \frac{2}{3}
    \]
    \[
        \therefore\quad \lim_{u\to2}\frac{\sqrt{4u+1}-3}{u-2} = \frac{2}{3}
    \]

    % ======================== item f ========================
    \item $\lim_{x \to 16} \dfrac{4 - \sqrt{x}}{16x - x^2}$

    \solucion

    Racionalizamos el numerador multiplicando por el conjugado $\frac{4+\sqrt{x}}{4+\sqrt{x}}$:
    \begin{align*}
        \frac{4-\sqrt{x}}{16x-x^2} \cdot \frac{4+\sqrt{x}}{4+\sqrt{x}}
        &= \frac{16-x}{(16x-x^2)(4+\sqrt{x})}
        = \frac{16-x}{x(16-x)(4+\sqrt{x})}
        = \frac{1}{x(4+\sqrt{x})}
    \end{align*}
    \[
        \lim_{x\to16}\frac{1}{x(4+\sqrt{x})} = \frac{1}{16(4+\sqrt{16})} = \frac{1}{16(4+4)} = \frac{1}{16 \cdot 8} = \frac{1}{128}
    \]
    \[
        \therefore\quad \lim_{x\to16}\frac{4-\sqrt{x}}{16x-x^2} = \frac{1}{128}
    \]

    % ======================== item g ========================
    \item $\lim_{t \to 0} \dfrac{\sqrt{1 + t} - \sqrt{1 - t}}{t}$

    \solucion

    Racionalizamos el numerador:
    \begin{align*}
        \frac{\sqrt{1+t}-\sqrt{1-t}}{t} \cdot \frac{\sqrt{1+t}+\sqrt{1-t}}{\sqrt{1+t}+\sqrt{1-t}}
        &= \frac{(1+t)-(1-t)}{t\!\left(\sqrt{1+t}+\sqrt{1-t}\right)} \\
        &= \frac{2t}{t\!\left(\sqrt{1+t}+\sqrt{1-t}\right)}
        = \frac{2}{\sqrt{1+t}+\sqrt{1-t}}
    \end{align*}
    \[
        \lim_{t\to0}\frac{2}{\sqrt{1+t}+\sqrt{1-t}} = \frac{2}{1+1} = \frac{2}{2} = 1
    \]
    \[
        \therefore\quad \lim_{t\to0}\frac{\sqrt{1+t}-\sqrt{1-t}}{t} = 1
    \]

    % ======================== item h ========================
    \item $\lim_{t \to 0} \left(\dfrac{1}{t} - \dfrac{1}{t^2 + t}\right)$

    \solucion
    \begin{align*}
        \frac{1}{t} - \frac{1}{t+t^2}
        &= \frac{t+t^2-t}{t(t+t^2)}
        = \frac{t^2}{t^2+t^3}
        = \frac{t^2}{t^2(1+t)}
        = \frac{1}{1+t}
    \end{align*}
    \[
        \lim_{t\to0}\frac{1}{1+t} = \frac{1}{1+0} = 1
    \]
    \[
        \therefore\quad \lim_{t\to0}\left(\frac{1}{t}-\frac{1}{t^2+t}\right) = 1
    \]

    % ======================== item i ========================
    \item $\lim_{h \to 0} \dfrac{(3 + h)^{-1} - 3^{-1}}{h}$

    \solucion
    \begin{align*}
        \frac{(3+h)^{-1}-3^{-1}}{h}
        &= \frac{\dfrac{1}{3+h}-\dfrac{1}{3}}{h}
        = \frac{\dfrac{3-(3+h)}{3(3+h)}}{h}
        = \frac{-h}{3(3+h)\cdot h}
        = \frac{-1}{3(3+h)}
    \end{align*}
    \[
        \lim_{h\to0}\frac{-1}{3(3+h)} = \frac{-1}{3(3+0)} = \frac{-1}{9}
    \]
    \[
        \therefore\quad \lim_{h\to0}\frac{(3+h)^{-1}-3^{-1}}{h} = -\frac{1}{9}
    \]
\end{enumerate}

\section[Límites notables]{Utilizando límites notables, calcular los siguientes límites.}

\begin{enumerate}[label=\alph*)]
    % ======================== item a ========================
    \item $\lim_{x \to 0} \dfrac{15sen(x)}{2x}$

    \solucion

    Recordando que $\displaystyle\lim_{x\to0}\frac{\operatorname{sen}(x)}{x} = 1$:
    \begin{align*}
        \lim_{x\to0}\frac{15\operatorname{sen}(x)}{2x}
        &= \frac{15}{2}\lim_{x\to0}\frac{\operatorname{sen}(x)}{x}
        = \frac{15}{2} \cdot 1 = \frac{15}{2}
    \end{align*}
    \[
        \therefore\quad \lim_{x\to0}\frac{15\operatorname{sen}(x)}{2x} = \frac{15}{2}
    \]

    % ======================== item b ========================
    \item $\lim_{x \to 0} \dfrac{sen(2x)}{sen(5x)}$

    \solucion
    \begin{align*}
        \lim_{x\to0}\frac{\operatorname{sen}(2x)}{\operatorname{sen}(5x)}
        &= \frac{\displaystyle\lim_{x\to0}\frac{\operatorname{sen}(2x)}{2x}\cdot 2x}{\displaystyle\lim_{x\to0}\frac{\operatorname{sen}(5x)}{5x}\cdot 5x}
        = \frac{\displaystyle\lim_{x\to0}\frac{\operatorname{sen}(2x)}{2x}}{\displaystyle\lim_{x\to0}\frac{\operatorname{sen}(5x)}{5x}} \cdot \frac{2x}{5x}
        = \frac{1}{1} \cdot \frac{2}{5} = \frac{2}{5}
    \end{align*}
    \[
        \therefore\quad \lim_{x\to0}\frac{\operatorname{sen}(2x)}{\operatorname{sen}(5x)} = \frac{2}{5}
    \]

    % ======================== item c ========================
    \item $\lim_{x\to\frac{\pi}{2}} \dfrac{x-\frac{\pi}{2}}{sen\left(x-\dfrac{\pi}{2}\right)}$

    \solucion

    Recordando que $\displaystyle\lim_{x\to0}\frac{\operatorname{sen}(x)}{x} = 1$, se tiene $\displaystyle\lim_{x\to0}\frac{x}{\operatorname{sen}(x)} = 1$. Entonces:
    \[
        \lim_{x\to\frac{\pi}{2}}\frac{x-\frac{\pi}{2}}{\operatorname{sen}\!\left(x-\frac{\pi}{2}\right)}
        = \left(\lim_{x\to0}\frac{\operatorname{sen}(x)}{x}\right)^{-1} = 1^{-1} = 1
    \]
    \[
        \therefore\quad \lim_{x\to\frac{\pi}{2}}\frac{x-\frac{\pi}{2}}{\operatorname{sen}\!\left(x-\frac{\pi}{2}\right)} = 1
    \]

    \newpage
    % ======================== item d ========================
    \item $\displaystyle \lim_{x \to 0} \frac{1-\cos(2x)}{x}$

    \solucion

    Usamos la identidad trigonométrica:
    \[
        (1-\cos(2x))(1+\cos(2x))=\operatorname{sen}^2(2x)
    \]
    de donde:
    \[
        1-\cos(2x)=\frac{\operatorname{sen}^2(2x)}{1+\cos(2x)}
    \]
    Sustituyendo en el límite:
    \begin{align*}
        \lim_{x\to0}\frac{1-\cos(2x)}{x}
        &= \lim_{x\to0}\frac{\operatorname{sen}^2(2x)}{x(1+\cos(2x))} \\[6pt]
        &= \lim_{x\to0}
            \left(\frac{\operatorname{sen}(2x)}{x}\right)
            \left(\frac{\operatorname{sen}(2x)}{1+\cos(2x)}\right) \\[6pt]
        &= \left(\lim_{x\to0}\frac{\operatorname{sen}(2x)}{x}\right)
            \left(\lim_{x\to0}\frac{\operatorname{sen}(2x)}{1+\cos(2x)}\right) \\[6pt]
        &= \left(2\lim_{x\to0}\frac{\operatorname{sen}(2x)}{2x}\right)
            \left(\frac{\operatorname{sen}(0)}{1+\cos(0)}\right) \\[6pt]
        &= (2 \cdot 1)\left(\frac{0}{2}\right) = 0
    \end{align*}
    \[
        \therefore\quad \lim_{x\to0}\frac{1-\cos(2x)}{x} = 0
    \]

    % ======================== item e ========================
    \item $\lim_{x \to 0} \left(1 - \dfrac{2x}{3}\right)^{\frac{1}{x}}$

    \solucion

    Sea $y = -\dfrac{2}{3}x$, entonces $x = -\dfrac{3}{2}y$. Cuando $x\to0$, $y\to0$:
    \begin{align*}
        \lim_{x\to0}\left(1-\frac{2}{3}x\right)^{\frac{1}{x}}
        &= \lim_{y\to0}(1+y)^{-\frac{3}{2y}}
        = \left[\lim_{y\to0}(1+y)^{\frac{1}{y}}\right]^{-\frac{3}{2}}
        = e^{-\frac{3}{2}}
        = \frac{1}{\sqrt{e^3}}
    \end{align*}
    \[
        \therefore\quad \lim_{x\to0}\left(1-\frac{2}{3}x\right)^{\frac{1}{x}} = \frac{1}{\sqrt{e^3}}
    \]

    \newpage
    % ======================== item f ========================
    \item $\lim_{x \to \infty} \left(1 + \dfrac{2}{x}\right)^x$

    \solucion

    Recordando que $\displaystyle\lim_{x\to\infty}\left(1+\frac{1}{x}\right)^x = e$.
    
    Sea $y = \dfrac{x}{2}$, entonces $x = 2y$. Cuando $x\to\infty$, $y\to\infty$:
    \begin{align*}
        \lim_{x\to\infty}\left(1+\frac{2}{x}\right)^x
        &= \lim_{y\to\infty}\left(1+\frac{1}{y}\right)^{2y}
        = \left[\lim_{y\to\infty}\left(1+\frac{1}{y}\right)^y\right]^2
        = e^2
    \end{align*}
    \[
        \therefore\quad \lim_{x\to\infty}\left(1+\frac{2}{x}\right)^x = e^2
    \]

    % ======================== item g ========================
    \item $\lim_{u \to \infty} \left(\dfrac{u - 1}{u + 2}\right)^u$

    \solucion

    Reescribimos $\dfrac{u-1}{u+2}$:
    \[
        \frac{u-1}{u+2} = \frac{u+2-3}{u+2} = 1 - \frac{3}{u+2}
    \]
    Sea $y = u+2$, entonces $u = y-2$. Cuando $u\to\infty$, $y\to\infty$:
    \begin{align*}
        \lim_{u\to\infty}\left(1-\frac{3}{u+2}\right)^u
        &= \lim_{y\to\infty}\left(1-\frac{3}{y}\right)^{y-2} \\[6pt]
        &= \lim_{y\to\infty}\frac{\left(1-\dfrac{3}{y}\right)^y}{\left(1-\dfrac{3}{y}\right)^2} \\[6pt]
        &= \frac{\displaystyle\lim_{y\to\infty}\left(1-\frac{3}{y}\right)^y}
            {\displaystyle\lim_{y\to\infty}\left(1-\frac{3}{y}\right)^2} \\[6pt]
        &= \frac{e^{-3}}{1^2} \\[6pt]
        &= \frac{1}{e^3}
    \end{align*}
    \[
        \therefore\quad \lim_{u\to\infty}\left(\frac{u-1}{u+2}\right)^u = \frac{1}{e^3}
    \]

    \[
        L = \lim_{u\to1}\frac{\ln u}{u-1} = \lim_{h\to0}\frac{\ln(1+h)}{h}

    \[
        L = \lim_{h\to0} \frac{1}{h}\ln(1+h) = \lim_{h\to0}\ln(1+h)^{\frac{1}{h}}

    \[
        L = \ln\left[\lim_{h\to0}(1+h)^{\frac{1}{h}}\right]
    \]

    Reconocemos el límite notable $\displaystyle\lim_{h\to0}(1+h)^{\frac{1}{h}} = e$, por lo tanto:

    \[
        L = \ln(e) = 1
    \]

    \[
        \therefore\quad \lim_{u\to1}\frac{\ln u}{u-1} = 1
    \]

% ======================== ACTIVIDAD 7 ========================
\section[Límites infinitos]{Determine cada uno de los siguientes límites infinitos.}

\begin{enumerate}[label=\alph*)]
    % ======================== item a ========================
    \item $\lim_{x \to -3} \dfrac{x + 2}{x + 3}$

    \solucion

    Analizamos el comportamiento cuando $x \to -3$.
    
    Evaluando el numerador y denominador:

    Cuando $x \to -3^-$: el numerador $x+2 \to -1$ (negativo) y el denominador $x+3 \to 0^-$ (negativo), entonces:
    \[
        \frac{\text{negativo}}{\text{negativo}} = +\infty
    \]

    Cuando $x \to -3^+$: el numerador $x+2 \to -1$ (negativo) y el denominador $x+3 \to 0^+$ (positivo), entonces:
    \[
        \frac{\text{negativo}}{\text{positivo}} = -\infty
    \]

    Luego, debido a que los límites laterales no coinciden, \textbf{no existe} el límite:
    \[
        \lim_{x\to-3}\frac{x+2}{x+3} \quad \text{no existe}
    \]

    % ======================== item b ========================
    \item $\lim_{x \to 1} \dfrac{2 - x}{(x - 1)^2}$

    \solucion

    El denominador $(x-1)^2 \geq 0$ siempre, y cuando $x \to 1$ resulta $(x-1)^2 \to 0^+$.

    Cuando $x \to 1^-$: el numerador $2-x \to 1 > 0$ (positivo) y $(x-1)^2 \to 0^+$ (positivo):
    \[
        \frac{\text{positivo}}{\text{positivo}} = +\infty
    \]

    Cuando $x \to 1^+$: el numerador $2-x \to 1 > 0$ (positivo) y $(x-1)^2 \to 0^+$ (positivo):
    \[
        \frac{\text{positivo}}{\text{positivo}} = +\infty
    \]

    Para este caso el límite existe:
    \[
        \lim_{x\to1}\frac{2-x}{(x-1)^2} = +\infty
    \]

    \newpage
    % ======================== item c ========================
    \item $\lim_{x \to 5^-} \dfrac{e^x}{(x - 5)^3}$

    \solucion

    Cuando $x \to 5^-$ se tiene $e^x \to e^5$ con $e^5 > 0$ (positivo).

    Analizando el denominador: $(x-5)^3$, con $x \to 5^-$ se tiene $(5^- - 5)^3 < 0$ (negativo).

    Por lo tanto:
    \[
        \frac{\text{positivo}}{\text{negativo}} = -\infty
    \]
    \[
        \lim_{x\to5^-}\frac{e^x}{(x-5)^3} = -\infty
    \]

    % ======================== item d ========================
    \item $\lim_{x \to 2^-} \dfrac{x^2 - 2x}{x^2 - 4x + 4}$

    \solucion

    Factorizamos numerador y denominador:
    \[
        \frac{x^2 - 2x}{x^2 - 4x + 4} = \frac{x(x-2)}{(x-2)^2} = \frac{x}{x-2}
    \]

    Analizamos el signo de cada factor cuando $x \to 2^-$ (es decir, $x < 2$ pero cerca de $2$):

    \begin{itemize}
        \item Numerador: $x \to 2$, luego $x > 0$ \quad (positivo)
        \item Denominador: $x - 2 \to 0^-$, luego $x - 2 < 0$ \quad (negativo)
    \end{itemize}

    Por lo tanto:
    \[
        \lim_{x\to2^-}\frac{x}{x-2} = \frac{\text{positivo}}{\text{negativo}} = -\infty
    \]
    \[
        \therefore\quad \lim_{x\to2^-}\frac{x^2-2x}{x^2-4x+4} = -\infty
    \]
\end{enumerate}

% ======================== ACTIVIDAD 8 ========================
\section[Límites que no existen]{Encuentre el límite o demuestre que no existe.}

\begin{enumerate}[label=\alph*)]
    % ======================== item a ========================
    \item $\lim_{x \to \infty} \dfrac{1 - x^2}{x^3 - x + 1}$

    \solucion

    Dividimos numerador y denominador entre la mayor potencia de $x$ en el denominador, que es $x^3$:
    \[
        \lim_{x\to\infty}\frac{1-x^2}{x^3-x+1}
        = \lim_{x\to\infty}\frac{\dfrac{1}{x^3} - \dfrac{1}{x}}{\,1 - \dfrac{1}{x^2} + \dfrac{1}{x^3}\,}
        = \frac{0 - 0}{1 - 0 + 0} = \frac{0}{1} = 0
    \]
    \[
        \therefore\quad \lim_{x\to\infty}\frac{1-x^2}{x^3-x+1} = 0
    \]

    % ======================== item b ========================
    \item $\lim_{x \to \infty} \dfrac{4x^3 + 6x^2 - 2}{2x^3 - 4x + 5}$

    \solucion

    Dividimos numerador y denominador entre la mayor potencia, $x^3$:
    \[
        \lim_{x\to\infty}\frac{4x^3+6x^2-2}{2x^3-4x+5}
        = \lim_{x\to\infty}\frac{4 + \dfrac{6}{x} - \dfrac{2}{x^3}}{2 - \dfrac{4}{x^2} + \dfrac{5}{x^3}}
        = \frac{4+0-0}{2-0+0} = \frac{4}{2} = 2
    \]
    \[
        \therefore\quad \lim_{x\to\infty}\frac{4x^3+6x^2-2}{2x^3-4x+5} = 2
    \]

    % ======================== item c ========================
    \item $\lim_{x \to \infty} \dfrac{x^2}{\sqrt{x^4 + 1}}$

    \solucion

    Dividimos numerador y denominador entre $x^2$. Como $x\to\infty$, $x > 0$,
    entonces $\sqrt{x^4} = x^2$, y por lo tanto $\dfrac{\sqrt{x^4+1}}{x^2} = \sqrt{\dfrac{x^4+1}{x^4}}$:

    \[
        \lim_{x\to\infty}\frac{x^2}{\sqrt{x^4+1}}
        = \lim_{x\to\infty}\frac{\dfrac{x^2}{x^2}}{\dfrac{\sqrt{x^4+1}}{x^2}}
        = \lim_{x\to\infty}\frac{1}{\sqrt{\dfrac{x^4+1}{x^4}}}
        = \lim_{x\to\infty}\frac{1}{\sqrt{1+\dfrac{1}{x^4}}}
        = \frac{1}{\sqrt{1+0}}
        = 1
    \]
    \[
        \therefore\quad \lim_{x\to\infty}\frac{x^2}{\sqrt{x^4+1}} = 1
    \]

    % ======================== item d ========================
    \item $\displaystyle \lim_{x \to -\infty} \frac{\sqrt{9x^6-x}}{x^3+1}$

    \solucion

    Factorizamos $x^6$ dentro de la raíz:

    \[
    \sqrt{9x^6-x}
    =
    \sqrt{x^6\left(9-\frac1{x^5}\right)}
    =
    |x^3|\sqrt{9-\frac1{x^5}}
    \]

    Como $x\to -\infty$, se tiene $x^3<0$, luego:

    \[
    |x^3|=-x^3
    \]

    Entonces:
    \begin{align*}
    \lim_{x\to-\infty}
    \frac{\sqrt{9x^6-x}}{x^3+1}
    &=
    \lim_{x\to-\infty}
    \frac{-x^3\sqrt{9-\frac1{x^5}}}{x^3+1}
    \\[6pt]
    &=
    \lim_{x\to-\infty}
    \frac{-\sqrt{9-\frac1{x^5}}}{1+\frac1{x^3}}
    \\[6pt]
    &=
    \frac{-\sqrt9}{1+0}
    \\[6pt]
    &=-3
    \end{align*}
    \[
    \therefore\quad
    \lim_{x\to-\infty}\frac{\sqrt{9x^6-x}}{x^3+1}=-3
    \]

    % ======================== item e ========================
    \item $\lim_{x \to \infty} \left(\sqrt{9x^2 + x} - 3x\right)$

    \solucion

    Multiplicamos por el conjugado $\dfrac{\sqrt{9x^2+x}+3x}{\sqrt{9x^2+x}+3x}$:
    \begin{align*}
        \left(\sqrt{9x^2+x}-3x\right)\cdot\frac{\sqrt{9x^2+x}+3x}{\sqrt{9x^2+x}+3x}
        &= \frac{(9x^2+x)-9x^2}{\sqrt{9x^2+x}+3x}
        = \frac{x}{\sqrt{9x^2+x}+3x}
    \end{align*}
    Dividimos por $x$ (con $x > 0$ para $x\to\infty$):
    \begin{align*}
        &= \frac{x}{x\left(\sqrt{9+\dfrac{1}{x}}+3\right)}
        = \frac{1}{\sqrt{9+\dfrac{1}{x}}+3}
    \end{align*}
    \[
        \lim_{x\to\infty}\frac{1}{\sqrt{9+\dfrac{1}{x}}+3}
        = \frac{1}{\sqrt{9+0}+3} = \frac{1}{3+3} = \frac{1}{6}
    \]
    \[
        \therefore\quad \lim_{x\to\infty}\left(\sqrt{9x^2+x}-3x\right) = \frac{1}{6}
    \]

    % ======================== item f ========================
    \item $\lim_{x \to \infty} \left(x + \sqrt{x^2 + 2x}\right)$

    \solucion

    Reescribimos $\sqrt{x^2+2x} = \sqrt{x^2\!\left(1+\dfrac{2}{x}\right)} = x\sqrt{1+\dfrac{2}{x}}$ (para $x > 0$):
    \begin{align*}
        \lim_{x\to\infty}\left(x+\sqrt{x^2+2x}\right)
        &= \lim_{x\to\infty}\left(x + x\sqrt{1+\frac{2}{x}}\right)
        = \lim_{x\to\infty} x\left(1+\sqrt{1+\frac{2}{x}}\right)
    \end{align*}
    Cuando $x\to\infty$: $\sqrt{1+\dfrac{2}{x}} \to 1$, por lo tanto:
    \[
        \lim_{x\to\infty} x\left(1+\sqrt{1+\frac{2}{x}}\right) = \lim_{x\to\infty} x \cdot (1+1) = 2\cdot\infty = +\infty
    \]
    \[
        \therefore\quad \lim_{x\to\infty}\left(x+\sqrt{x^2+2x}\right) = +\infty
    \]
\end{enumerate}
