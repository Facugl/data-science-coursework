\section[Dominio de funciones]{Determine el dominio más amplio (de números reales) en donde la expresión algebraica dada defina una función real.}

\begin{enumerate}[label=\alph*.-]
    % item a.-
    \item $f(x) = x^2 + 1$
    
    \textbf{Solución:}
    
    En este caso el dominio de $f$, el cual denotaremos con $D_f$, es: $D_f = \mathbb{R}$.

    Expresado en notación de intervalo sería: $D_f = (-\infty, \infty)$.
    
    % item b.-
    \item $g(x) = \sqrt{3x - 2}$
    
    \textbf{Solución:}
    
    La raíz cuadrada de un número negativo no está definida en $\mathbb{R}$. Entonces, debe ser:
    \[3x - 2 \geq 0\]
    \[x \geq \frac{2}{3}\]
    
    Luego $D_g = \mathbb{R} \mid x \geq \frac{2}{3}$
    
    Expresado en notación de intervalo sería: $D_g = \left[\frac{2}{3}, \infty\right)$.
    
    % item c.-
    \item $h(x) = \sqrt{16 - x^2}$
    
    \textbf{Solución:}
    
    Tenemos que:
    \[16 - x^2 \geq 0\]
    \[x^2 \leq 16\]
    \[|x| \leq 4\]
    
    Luego $D_h = \mathbb{R} \mid |x| \leq 4$
    
    Expresado en intervalo: $D_h = [-4, 4]$
    
    \newpage

    % item d.-
    \item $u(x) = \dfrac{x+4}{4-x^2}$
    
    \textbf{Solución:}
    
    En este caso el denominador debe ser $\neq 0$ ya que en $\mathbb{R}$ la división por 0 no está definida.
    
    Entonces:
    \[4 - x^2 \neq 0\]
    \[x^2 \neq 4 \implies x \neq 2 \quad \text{y} \quad x \neq -2\]
    
    Entonces $D_u = \mathbb{R} - \{-2, 2\}$
    
    En intervalo: $D_u = (-\infty, -2) \cup (-2, 2) \cup (2, \infty)$
    
    % item e.-
    \item $v(t) = \sqrt{1 + 2t^2}$
    
    \textbf{Solución:}
    
    El radicando debe ser $\geq 0$
    
    Tenemos $t^2$, el cual siempre es $\geq 0$

    Luego $D_e = \mathbb{R}$.
    
    En notación de intervalo $D_e = (-\infty, \infty)$
    
    % item f.-
    \item $w(x) = \dfrac{3x}{\sqrt{x+1}}$
    
    \textbf{Solución:}
    
    Tenemos que:
    \[\sqrt{x+1} \neq 0\]
    \[x + 1 \neq 0\]
    \[x \neq -1\]
    
    Pero además tenemos que para $x \leq -1$, $\sqrt{x+1}$ no está definida en $\mathbb{R}$. Entonces:
    
    \[\sqrt{x+1} > 0\]
    \[x > -1\]
    
    Luego el dominio se reduce a: $D_w = \mathbb{R} \mid x > -1$
    
    En intervalo: $D_w = (-1, \infty)$
    
    \newpage
    
    % item g.-
    \item $s(t) = \dfrac{\sqrt{2+t}}{3-t}$
    
    \textbf{Solución:}
    
    Buscando las restricciones para este caso tenemos:
    
    1. Para:
    \[2 + t \geq 0\]
    \[t \geq -2\]
    
    2. Denominador:
    \[3 - t \neq 0\]
    \[t \neq 3\]
    
    Por lo tanto el $D_s = \mathbb{R} \mid t \geq -2 - \{3\}$
    
    En notación de intervalo: $D_s = [-2, 3) \cup (3, \infty)$
    
    % item g.-
    \item $t(\theta) = \sqrt[3]{3\theta - 1}$
    
    \textbf{Solución:}
    
    En este caso no tenemos ninguna restricción relacionada con la raíz. El radicando puede ser $+$ o $-$.
    
    Tenemos que el $D_t = \mathbb{R}$
    
    En notación de intervalo: $D_t = (-\infty, \infty)$
\end{enumerate}

\section[Transformaciones de funciones]{A partir de la gráfica de la función $f(x) = x^2$ y mediante transformaciones, dibuje la gráfica de las funciones:}

\begin{enumerate}[label=\alph*.-]
    \item $g(x) = x^2 - 1$
    \item $h(x) = (x - 1)^2$
    \item $m(x) = (x + 1)^2 - 2$
    \item $q(x) = -2(x - 1)^2 + 1$
\end{enumerate}

\textbf{Solución:}
\begin{enumerate}[label=\alph*.-]
    % item a.-
    \item $g(x) = x^2 - 1$: Se desplaza $-1$ unidad en el eje $y$.
    
    % item b.-
    \item $h(x) = (x-1)^2$: Se desplaza $1$ unidad en el eje $x$.
    
    % item c.-
    \item $m(x) = (x+1)^2 - 2$: Se desplaza $-1$ unidad en el eje $x$. Luego se desplaza $-2$ unidades en el eje $y$.
    
    % item d.-
    \item $q(x) = -2(x-1)^2 + 1$: Se desplaza $1$ unidad en el eje $x$. Luego sufre un estiramiento vertical por un factor de 2 unidades. Luego dado el signo $-$ se refleja en $x$. Finalmente se desplaza $1$ unidad en el eje $y$.
\end{enumerate}

\section[Identificación de transformaciones]{Se da la gráfica de $y = f(x)$. Identifique la gráfica de cada una de las siguientes funciones obtenidas a partir de transformaciones de $f$.}

\begin{enumerate}[label=\alph*.-]
    \item $f_1(x) = f(x - 4)$
    \item $f_2(x) = f(x) + 3$   
    \item $f_3(x) = \dfrac{1}{3}f(x)$
    \item $f_4(x) = -f(x + 4)$
    \item $f_5(x) = 2f(x + 6)$
\end{enumerate}

\textbf{Solución:}

Basándonos en las transformaciones elementales de funciones:

\begin{enumerate}[label=\alph*.-]
    \item $f_1(x) = f(x-4)$: Desplazamiento horizontal de 4 unidades en el eje $x$. Corresponde a la gráfica identificada como $f_1$ en el diagrama.
    \item $f_2(x) = f(x) + 3$: Desplazamiento vertical de 3 unidades en el eje $y$. Corresponde a $f_2$ en el diagrama.
    \item $f_3(x) = \frac{1}{3}f(x)$: Compresión vertical (factor $\frac{1}{3}$). La gráfica se aplana un tercio de su altura original. Corresponde a $f_3$ en el diagrama.
    \item $f_4(x) = -f(x+4)$: Primero desplazamiento horizontal de 4 unidades en el eje $x$, luego reflexión sobre el eje $x$. Corresponde a $f_4$ en el diagrama.
    \item $f_5(x) = 2f(x+6)$: Desplazamiento horizontal de 6 unidades en el eje $x$, seguido de un estiramiento vertical (factor 2). Corresponde a $f_5$ en el diagrama.
\end{enumerate}

\section[Análisis gráfico de funciones]{Trace la gráfica de las siguientes funciones y para cada función determine:}
 
\begin{enumerate}[label=\roman*.]
    \item Dominio e imagen.
    \item Intervalos de crecimiento y decrecimiento.
\end{enumerate}
 
\begin{enumerate}[label=\alph*.-]
    % item a.-
    \item $f(x) = x^2 - 5x$
    
    \textbf{Solución:}
    
    Tenemos que $f(x) = x^2 - 5x$ es una función cuadrática con $a = 1$, $b = -5$ y $c = 0$. Como $a > 0$, es cóncava hacia arriba.
    
    Calculamos el vértice $V = (x_v, y_v)$:
    \[
    x_v = -\frac{b}{2a} = -\frac{(-5)}{2 \cdot 1} = \frac{5}{2}
    \]
    \[
    y_v = f(x_v) = \left(\frac{5}{2}\right)^2 - 5 \cdot \frac{5}{2} = \frac{25}{4} - \frac{25}{2} = -\frac{25}{4}
    \]
    \[
    V = \left(\frac{5}{2},\ -\frac{25}{4}\right)
    \]
 
    Intersección con el eje $x$ (haciendo $f(x) = 0$):
    \[
    0 = x^2 - 5x = x(x - 5) \implies x = 0 \quad \land \quad x = 5
    \]
    Puntos: $P = (0, 0)$ y $Q = (5, 0)$.
 
    Intersección con el eje $y$ (haciendo $x = 0$): $C = (0, 0)$.

    \begin{enumerate}[label=\roman*.]
        \item $D_f = \mathbb{R}$ \quad e \quad $I_f = \left[-\dfrac{25}{4},\ +\infty\right)$
        \item $f(x)$ es \textbf{decreciente} en $\left(-\infty,\ \dfrac{5}{2}\right)$ y \textbf{creciente} en $\left(\dfrac{5}{2},\ +\infty\right)$.
    \end{enumerate}
 
    % item b.-
    \item $g(x) = 2x^2 + 2x - 12$
    
    \textbf{Solución:}
    
    Tenemos $a = 2$, $b = 2$, $c = -12$. Como $a > 0$, $g(x)$ es cóncava hacia arriba.
 
    Vértice:
    \[
    x_v = -\frac{b}{2a} = -\frac{2}{2 \cdot 2} = -\frac{1}{2}
    \]
    \[
    y_v = g\!\left(-\tfrac{1}{2}\right) = 2\!\left(-\tfrac{1}{2}\right)^2 + 2\!\left(-\tfrac{1}{2}\right) - 12 = \frac{1}{2} - 1 - 12 = -\frac{25}{2}
    \]
    \[
    V = \left(-\frac{1}{2},\ -\frac{25}{2}\right)
    \]
 
    Intersección con el eje $x$ (fórmula cuadrática):
    \[
    x_{1,2} = \frac{-2 \pm \sqrt{4 - 4 \cdot 2 \cdot (-12)}}{4} = \frac{-2 \pm \sqrt{100}}{4} = \frac{-2 \pm 10}{4}
    \implies x_1 = 2,\quad x_2 = -3
    \]
    Puntos: $P = (2, 0)$ y $Q = (-3, 0)$.
 
    Intersección con el eje $y$: $c = g(0) = -12$, luego $C = (0, -12)$.

\section[Operaciones con funciones]{Para los siguientes pares de funciones, encuentre:}
 
\begin{enumerate}[label=\roman*.]
    \item $(f \circ g)(x)$
    \item $(g \circ f)(x)$
    \item $(f \pm g)(x)$
    \item $(f \cdot g)(x)$
    \item $(f / g)(x)$
\end{enumerate}
 
En todos los casos indique el dominio.
 
\begin{enumerate}[label=\alph*.-]
 
    % item a.-
    \item $f(x) = \dfrac{1}{x}$ \quad y \quad $g(x) = 2x + 4$
    
    \textbf{Solución:}
 
    \begin{enumerate}[label=\roman*.]
        \item \textbf{$(f \circ g)(x) = f(g(x))$}
        
        $D_g = \mathbb{R}$. Entonces:
        \[
        f(g(x)) = \frac{1}{2x + 4}
        \]
        Para que esté definida: $2x + 4 \neq 0 \implies x \neq -2$.
        
        $D_{f \circ g} = \mathbb{R} - \{-2\}$, es decir $(-\infty, -2) \cup (-2, +\infty)$.
 
        \item \textbf{$(g \circ f)(x) = g(f(x))$}
        
        $D_f = \mathbb{R} - \{0\}$. Entonces:
        \[
        g(f(x)) = 2\cdot\frac{1}{x} + 4 = \frac{2}{x} + 4
        \]
        La restricción proviene de $f$: $x \neq 0$.
        
        $D_{g \circ f} = \mathbb{R} - \{0\}$, es decir $(-\infty, 0) \cup (0, +\infty)$.
 
        \item \textbf{$(f \pm g)(x) = f(x) \pm g(x)$}
        \[
        (f \pm g)(x) = \frac{1}{x} \pm (2x + 4)
        \]
        Restricción: $x \neq 0$ (por $f$). $\quad D_{f \pm g} = \mathbb{R} - \{0\}$.
 
        \item \textbf{$(f \cdot g)(x) = f(x) \cdot g(x)$}
        \[
        (f \cdot g)(x) = \frac{1}{x}(2x + 4) = 2 + \frac{4}{x}
        \]
        Restricción: $x \neq 0$. $\quad D_{f \cdot g} = \mathbb{R} - \{0\}$.
 
        \item \textbf{$(f/g)(x) = \dfrac{f(x)}{g(x)}$}
        \[
        (f/g)(x) = \frac{\dfrac{1}{x}}{2x + 4} = \frac{1}{x(2x+4)} = \frac{1}{2x^2 + 4x}
        \]
        Restricciones: $x \neq 0$ y $2x + 4 \neq 0 \implies x \neq -2$.
        
        $D_{f/g} = \mathbb{R} - \{-2,\ 0\}$, es decir $(-\infty, -2) \cup (-2, 0) \cup (0, +\infty)$.
    \end{enumerate}
 
    % item b.-
    \item $f(x) = x^2$ \quad y \quad $g(x) = \sqrt{x - 3}$
    
    \textbf{Solución:}
 
    \begin{enumerate}[label=\roman*.]
        \item \textbf{$(f \circ g)(x) = f(g(x))$}
        
        $D_g = [3, +\infty)$ (pues $x - 3 \geq 0 \implies x \geq 3$). Entonces:
        \[
        f(g(x)) = \left(\sqrt{x-3}\right)^2 = x - 3
        \]
        $D_{f \circ g} = [3, +\infty)$.
 
        \item \textbf{$(g \circ f)(x) = g(f(x))$}
        
        $D_f = \mathbb{R}$. Entonces:
        \[
        g(f(x)) = \sqrt{x^2 - 3}
        \]
        Para que esté definida: $x^2 - 3 \geq 0 \implies x^2 \geq 3 \implies |x| \geq \sqrt{3}$.
        
        $D_{g \circ f} = \left(-\infty, -\sqrt{3}\right] \cup \left[\sqrt{3}, +\infty\right)$.
 
        \item \textbf{$(f \pm g)(x) = f(x) \pm g(x)$}
        \[
        (f \pm g)(x) = x^2 \pm \sqrt{x - 3}
        \]
        Restricción: $x \geq 3$ (por $g$). $\quad D_{f \pm g} = [3, +\infty)$.
 
        \item \textbf{$(f \cdot g)(x) = f(x) \cdot g(x)$}
        \[
        (f \cdot g)(x) = x^2 \cdot \sqrt{x - 3}
        \]
        $D_{f \cdot g} = [3, +\infty)$.
 
        \item \textbf{$(f/g)(x) = \dfrac{f(x)}{g(x)}$}
        \[
        (f/g)(x) = \frac{x^2}{\sqrt{x-3}}
        \]
        Restricciones: $x - 3 \geq 0$ y $\sqrt{x-3} \neq 0 \implies x > 3$.
        
        $D_{f/g} = D_f \cap D_g \setminus \{3\} = (3, +\infty)$.
    \end{enumerate}
\end{enumerate}

    \begin{enumerate}[label=\roman*.]
        \item El costo total para una empresa es la suma del costo variable y los costos fijos. Escriba el costo total $C$ como función del número de unidades producidas.
        \item Escriba el ingreso $R$ como función del número de unidades vendidas.
        \item Escriba la utilidad $P$ como función del número de unidades vendidas.
        
        (Nota: $P = R - C$).
    \end{enumerate}
    
    \textbf{Solución:}
    
    \begin{enumerate}[label=\roman*.]
        % item a.-
        \item \textbf{Costo Total $C_T(x)$}
        
        El costo total es la suma del costo variable y los costos fijos:
        \[C_T(x) = C_{\text{fijo}} + C_{\text{variable}}\]
        
        Donde el costo variable es $\$12.30$ por unidad multiplicado por $x$ unidades:
        \[C_T(x) = 98000 + 12.30x\]
        
        Por lo tanto: \[\boxed{C_T(x) = 12.30x + 98000}\]
        
        \item \textbf{Ingreso $R(x)$}
        
        El ingreso se obtiene multiplicando el precio de venta por el número de unidades vendidas:
        \[R(x) = 17.98 \cdot x\]
        
        Por lo tanto: \[\boxed{R(x) = 17.98x}\]
        
        \item \textbf{Utilidad $P(x)$}
        
        La utilidad es la diferencia entre el ingreso y el costo total:
        \[P(x) = R(x) - C_T(x)\]
        \[P(x) = 17.98x - (12.30x + 98000)\]
        \[P(x) = 17.98x - 12.30x - 98000\]
        \[P(x) = 5.68x - 98000\]
        
        Por lo tanto: \[\boxed{P(x) = 5.68x - 98000}\]
    \end{enumerate}
    
    % item b.-
    \item La función que relaciona la temperatura medida en grados Celsius con la temperatura en Kelvins, es lineal.

        \item Exprese el punto de ebullición 100°C, en Kelvins.
        \item Se define el cero absoluto como 0K. ¿Cuántos es 0K en grados Celsius?
    \end{enumerate}
    
    \textbf{Solución:}
    
    \begin{enumerate}[label=\roman*.]
        \item \textbf{Función de conversión}
        
        Sabemos que la relación es lineal, de la forma:
        \[K = m \cdot C + b\]
        
        Tenemos dos puntos conocidos:
        \begin{itemize}
            \item Punto 1: $(0C, 273K)$
            \item Punto 2: $(27C, 300K)$
        \end{itemize}
        
        Calculamos la pendiente $m$:
        \[m = \frac{300 - 273}{27 - 0} = \frac{27}{27} = 1\]
        
        Ahora encontramos $b$ usando el punto $(0C, 273K)$:
        \[273 = 1 \cdot 0 + b\]
        \[b = 273\]
        
        Por lo tanto, la función de conversión es:
        \[\boxed{K = C + 273} \quad \text{o bien} \quad \boxed{f(C) = C + 273}\]
        
        \item \textbf{Punto de ebullición en kelvins}
        
        Aplicamos la función con $C = 100$:
        \[K = 100 + 273 = 373\]
        
        Por lo tanto: \[\boxed{100C = 373K}\]
        
        \item \textbf{Cero absoluto en grados Celsius}
        
        Sabemos que $K = C + 273$. Si $K = 0$:
        \[0 = C + 273\]
        \[C = -273\]
        
        Por lo tanto: \[\boxed{0K = -273C}\]
    \end{enumerate}

\section[Modelado exponencial]{Resuelve las siguientes situaciones problemáticas:}

\begin{enumerate}[label=\alph*.-]
    \item Cierto cultivo de la bacteria Streptococcus A inicialmente tiene 10 bacterias y se observa que se duplica cada 1,5 horas.
    \begin{enumerate}[label=\roman*.]
        \item Encuentre un modelo exponencial $n(t) = n_0 2^{t/a}$ para el número de bacterias en el cultivo después de $t$ horas.
        \item Estime el número de bacterias después de 35 horas.
        \item ¿Cuándo llegará a 10,000 el número de bacterias?
    \end{enumerate}
    
    \textbf{Solución:}
    
    % item a.-
    \begin{enumerate}[label=\roman*.]
        \item \textbf{Encontrar el modelo exponencial}
        
        El modelo general para crecimiento exponencial con duplicación es:
        \[n(t) = n_0 \cdot 2^{t/a}\]
        
        Donde:
        \begin{itemize}
            \item $n_0$ = número inicial de bacterias
            \item $t$ = tiempo transcurrido (en horas)
            \item $a$ = tiempo de duplicación (tiempo que tarda en duplicarse)
        \end{itemize}
        
        Datos del problema:
        \begin{itemize}
            \item $n_0 = 10$ bacterias (población inicial)
            \item $a = 1.5$ horas (se duplica cada 1.5 horas)
        \end{itemize}
        
        Sustituyendo en el modelo:
        \[\boxed{n(t) = 10 \cdot 2^{t/1.5}}\]
        
        \item \textbf{Población después de 35 horas}
        
        Sustituimos $t = 35$ en nuestro modelo:
        \[n(35) = 10 \cdot 2^{35/1.5}\]
        
        Calculamos el exponente:
        \[\frac{35}{1.5} = 23.33\]
        
        Por lo tanto:
        \[n(35) = 10 \cdot 2^{23.33}\]
        
        Calculamos $2^{23.33}$:
        \[2^{23.33} \approx 10,321,285\]
        
        \[n(35) = 10 \cdot 10,321,285 \approx 103,212,850 \text{ bacterias}\]
        
        \[\boxed{n(35) \approx 103,212,850 \text{ bacterias}}\]
        
        \item \textbf{Tiempo para llegar a 10,000 bacterias}

        Planteamos:
        \[10,000 = 10 \cdot 2^{t/1.5}\]

        Dividimos entre 10:
        \[1,000 = 2^{t/1.5}\]

        Aplicamos logaritmo natural en ambos lados:
        \[\ln(1,000) = \ln\left(2^{t/1.5}\right)\]

        Usando la propiedad $\ln(a^b) = b \ln(a)$:
        \[\ln(1,000) = \frac{t}{1.5} \ln(2)\]

        Despejamos $t$:
        \[t = \frac{1.5 \ln(1,000)}{\ln(2)}\]

        Calculamos:
        \[\ln(1,000) \approx 6.907\]
        \[\ln(2) \approx 0.693\]

        \[t = \frac{1.5 \cdot 6.907}{0.693} \approx 14.95 \text{ horas}\]

        \[\boxed{t \approx 15 \text{ horas}}\]
        
    \end{enumerate}
    
    \newpage
    
    % item b.-
    \item La población de zorros en cierta región tiene una tasa de crecimiento relativa de 8\% por año. Se estima que la población en 2005 era de 18,000.
    \begin{enumerate}[label=\roman*.]
        \item Encuentre una función $n(t) = n_0 e^{rt}$ que modele la población en años después de 2005.
        \item Use la función del inciso i. para estimar la población de zorros en el año 2013.
        \item Trace una gráfica de la función de población de zorros para los años 2005-2013.
    \end{enumerate}
    
    \textbf{Solución:}
    
    \begin{enumerate}[label=\roman*.]
        \item \textbf{Encontrar el modelo exponencial}
        
        El modelo general para crecimiento continuo es:
        \[n(t) = n_0 \cdot e^{rt}\]
        
        Donde:
        \begin{itemize}
            \item $n_0$ = población inicial
            \item $r$ = tasa de crecimiento relativa (como decimal)
            \item $t$ = tiempo transcurrido (en años)
            \item $e \approx 2.718$ (número de Euler)
        \end{itemize}
        
        Datos del problema:
        \begin{itemize}
            \item $n_0 = 18,000$ (población en 2005)
            \item $r = 0.08$ (8\% = 8/100 = 0.08)
            \item $t$ = años después de 2005
        \end{itemize}
        
        Sustituyendo en el modelo:
        \[\boxed{n(t) = 18,000 \cdot e^{0.08t}}\]
        
        \item \textbf{Población en el año 2013}
        
        El año 2013 está a $t = 2013 - 2005 = 8$ años de 2005.
        
        Sustituimos $t = 8$ en nuestro modelo:
        \[n(8) = 18,000 \cdot e^{0.08 \cdot 8}\]
        
        Calculamos el exponente:
        \[0.08 \times 8 = 0.64\]
        
        \[n(8) = 18,000 \cdot e^{0.64}\]
        
        Calculamos $e^{0.64}$:
        \[e^{0.64} \approx 1.896\]
        
        \[n(8) = 18,000 \times 1.896 \approx 34,128 \text{ zorros}\]

    \item El yodo radiactivo $^{131}I$, que se usa con frecuencia en estudios de exploración de la glándula tiroides, se desintegra según la fórmula $N = N_0(0.5)^{t/8}$ donde $N_0$ es la dosis inicial y $t$ es el tiempo en días.
    \begin{enumerate}[label=\roman*.]
        \item Si la dosis inicial es de 64 gramos de yodo radiactivo, ¿Cuántos gramos estarán presentes después de 4 días?
        \item ¿Cuál es la vida media del $^{131}I$?
        \item ¿Cuánto tiempo tienen que transcurrir para que queden 2 gramos?
    \end{enumerate}
    
    \textbf{Solución:}
    
    El modelo de desintegración es: $N(t) = N_0(0.5)^{t/8}$

    \begin{enumerate}[label=\roman*.]
        \item \textbf{Cantidad después de 4 días}
        
        Datos:
        \begin{itemize}
            \item $N_0 = 64$ gramos (dosis inicial)
            \item $t = 4$ días
        \end{itemize}
        
        Sustituimos en la fórmula:
        \[N(4) = 64 \cdot (0.5)^{4/8}\]
        
        Simplificamos el exponente:
        \[\frac{4}{8} = 0.5\]
        
        \[N(4) = 64 \cdot (0.5)^{0.5}\]
        
        Recordemos que $(0.5)^{0.5} = \sqrt{0.5}$:
        \[N(4) = 64 \cdot \sqrt{0.5} = 64 \cdot 0.707 \approx 45.25 \text{ gramos}\]
        
        \[\boxed{N(4) \approx 45.25 \text{ gramos}}\]
        
        \newpage
        \item \textbf{Vida media del $^{131}I$}
        
        La vida media es el tiempo en que una sustancia radiactiva se reduce a la mitad de su cantidad inicial.
        
        En la fórmula general $N = N_0(0.5)^{t/h}$, el valor $h$ en el denominador del exponente representa la vida media.
        
        Comparando con nuestra fórmula $N = N_0(0.5)^{t/8}$:
        
        \[\boxed{\text{La vida media es } h = 8 \text{ días}}\]
        
        \textit{Verificación:} Si $t = 8$ días:
        \[N(8) = N_0(0.5)^{8/8} = N_0(0.5)^1 = 0.5N_0\]
        Efectivamente, se reduce a la mitad. \checkmark
        
        \item \textbf{Tiempo para que queden 2 gramos}
        
        Queremos encontrar $t$ cuando $N(t) = 2$ gramos, con $N_0 = 64$ gramos.
        
        Planteamos la ecuación:
        \[2 = 64 \cdot (0.5)^{t/8}\]
        
        Dividimos ambos lados entre 64:
        \[\frac{2}{64} = (0.5)^{t/8}\]
        
        Simplificamos:
        \[\frac{1}{32} = (0.5)^{t/8}\]
        
        Observamos que $\frac{1}{32} = \frac{1}{2^5} = (0.5)^5$
        
        Por lo tanto:
        \[(0.5)^5 = (0.5)^{t/8}\]
        
        Como las bases son iguales, los exponentes deben ser iguales:
        \[5 = \frac{t}{8}\]
        
        Despejamos $t$:
        \[t = 5 \times 8 = 40 \text{ días}\]
        
        \[\boxed{t = 40 \text{ días}}\]
        
        \textit{Verificación usando logaritmos:}
        
        Partiendo de $\frac{1}{32} = (0.5)^{t/8}$, aplicamos logaritmo en ambos lados:
        \[\log\left(\frac{1}{32}\right) = \frac{t}{8} \cdot \log(0.5)\]
        
        Calculamos:
        \[\log(1/32) = -\log(32) = -1.505\]
        \[\log(0.5) = -0.301\]
        
        \[\frac{t}{8} = \frac{-1.505}{-0.301} = 5\]
        \[t = 40 \text{ días}\]
        
        Para pasar de 64 gr a 2 gr necesitamos 5 reducciones a la mitad:
        \[64 \xrightarrow{\div 2} 32 \xrightarrow{\div 2} 16 \xrightarrow{\div 2} 8 \xrightarrow{\div 2} 4 \xrightarrow{\div 2} 2\]
        Como cada reducción toma 8 días: $5 \times 8 = 40$ días.
    \end{enumerate}

    \item La vida media del radio 226 es de 1600 años. Suponga que tenemos una muestra de 22 mg.
    \begin{enumerate}[label=\roman*.]
        \item Encuentre una función $m(t) = m_0 2^{-t/h}$ que modele la masa restante después de $t$ años.
        \item Encuentre una función $m(t) = m_0 e^{-rt}$ que modele la masa restante después de $t$ años.
        \item ¿Cuánto de la muestra habrá después de 4000 años?
        \item ¿Después de cuánto tiempo habrá sólo 18 mg de la muestra?
    \end{enumerate}
    
    \textbf{Solución:}
    
    \begin{enumerate}[label=\roman*.]
        \item \textbf{Modelo con base 2}
        
        El modelo general para decaimiento radiactivo con base 2 es:
        \[m(t) = m_0 \cdot 2^{-t/h}\]
        
        Donde:
        \begin{itemize}
            \item $m_0$ = masa inicial
            \item $t$ = tiempo transcurrido (en años)
            \item $h$ = vida media (tiempo para reducirse a la mitad)
            \item El exponente negativo indica decaimiento
        \end{itemize}
        
        Datos:
        \begin{itemize}
            \item $m_0 = 22$ mg
            \item $h = 1600$ años
        \end{itemize}
        
        \[\boxed{m(t) = 22 \cdot 2^{-t/1600}} \quad \text{($t$ en años)}\]
        
        \newpage

        \item \textbf{Modelo con base $e$}
        
        El modelo exponencial natural es:
        \[m(t) = m_0 \cdot e^{-rt}\]
        
        Para convertir del modelo base 2 al modelo base $e$, usamos la relación:
        \[2^{-t/h} = e^{-rt} \quad \text{donde } r = \frac{\ln 2}{h}\]
        
        Sabemos que $2 = e^{\ln 2}$, entonces:
        \[2^{-t/h} = \left(e^{\ln 2}\right)^{-t/h} = e^{(-t/h)\ln 2} = e^{-t(\ln 2/h)}\]
        
        Por lo tanto: $r = \dfrac{\ln 2}{h}$
        
        Calculamos $r$:
        \[r = \frac{\ln 2}{h} = \frac{0.6931}{1600} \approx 0.000433\]
        
        \[\boxed{m(t) = 22 \cdot e^{-0.000433t}} \quad \text{($t$ en años)}\]
        
        Ambos modelos son equivalentes, solo cambia la base del exponencial ($2$ vs $e$).
        
        \item \textbf{Masa después de 4000 años}
        
        Usamos el modelo base 2 (es más fácil para este cálculo):
        \[m(4000) = 22 \cdot 2^{-4000/1600}\]
        
        Simplificamos el exponente:
        \[\frac{4000}{1600} = 2.5\]
        
        \[m(4000) = 22 \cdot 2^{-2.5}\]
        
        \[m(4000) = 22 \times 0.177 \approx 3.89 \text{ mg}\]
        
        \[\boxed{m(4000) \approx 3.89 \text{ mg}}\]
        
        \textit{Verificación con el modelo base $e$:}
        \[m(4000) = 22 \cdot e^{-0.000433 \times 4000} = 22 \cdot e^{-1.732}\]
        \[e^{-1.732} \approx 0.177\]
        \[m(4000) = 22 \times 0.177 \approx 3.89 \text{ mg}\]
        
        \item \textbf{Tiempo para tener 18 mg}
        
        Queremos encontrar $t$ cuando $m(t) = 18$ mg.
        
        Usamos el modelo base $e$:
        \[18 = 22 \cdot e^{-rt}\]
        
        Donde $r = 0.000433$.
        
        Dividimos entre 22:
        \[\frac{18}{22} = e^{-rt}\]
        
        Simplificamos:
        \[\frac{9}{11} = e^{-0.000433t}\]
        
        Aplicamos logaritmo natural en ambos lados:
        \[\ln\left(\frac{9}{11}\right) = \ln\left(e^{-0.000433t}\right)\]
        
        Por la propiedad $\ln(e^x) = x$:
        \[\ln\left(\frac{9}{11}\right) = -0.000433t\]
        
        Calculamos el logaritmo:
        \[\ln(9/11) = \ln(0.818) \approx -0.201\]
        
        Despejamos $t$:
        \[t = \frac{-0.201}{-0.000433} = \frac{0.201}{0.000433} \approx 464.2 \text{ años}\]
        
        \[\boxed{t \approx 464.2 \text{ años}}\]
        
    \end{enumerate}
\end{enumerate}
